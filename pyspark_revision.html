RDD Transformations (Lazy)
rdd.map(lambda x: x * 2)              # 1-to-1 mapping
rdd.flatMap(lambda x: x.split(" "))    # 1-to-many mapping
rdd.filter(lambda x: x > 3)           # keep matching rows
rdd.distinct()                         # remove duplicates
rdd.sortBy(lambda x: x[1])            # sort by key func
rdd.groupByKey()                       # group (key,val) pairs
rdd.reduceByKey(lambda a,b: a+b)      # aggregate by key ✓ preferred
rdd.mapValues(lambda v: v*2)          # map only values
rdd.union(rdd2)                        # concatenate two RDDs
rdd.join(rdd2)                         # inner join on key
rdd.repartition(8)                    # reshuffle to N partitions
rdd.coalesce(2)                       # reduce partitions (no shuffle)
RDD Actions (Trigger Execution)
rdd.collect()          # return ALL data to driver ⚠️ use with care
rdd.count()            # count elements
rdd.first()            # first element
rdd.take(5)            # first N elements
rdd.top(3)             # largest N elements
rdd.reduce(lambda a,b: a+b)  # reduce to single value
rdd.sum()              # sum of elements
rdd.mean()             # average
rdd.max() / rdd.min() # max / min
rdd.countByKey()       # count per key (key,val) RDDs
rdd.saveAsTextFile("path") # write to disk
rdd.foreach(lambda x: print(x)) # execute func on each

Window Functions
from pyspark.sql.window import Window
import pyspark.sql.functions as F

w = Window.partitionBy("dept") \
           .orderBy("salary")

# Ranking
df.withColumn("rank", F.rank().over(w))
df.withColumn("dense", F.dense_rank().over(w))
df.withColumn("row_n", F.row_number().over(w))

# Lag / Lead
df.withColumn("prev", F.lag("salary", 1).over(w))
df.withColumn("next", F.lead("salary", 1).over(w))

# Running total
w2 = Window.partitionBy("dept") \
            .orderBy("date") \
            .rowsBetween(Window.unboundedPreceding, 0)
df.withColumn("running", F.sum("sales").over(w2))
UDFs
from pyspark.sql.functions import udf
from pyspark.sql.types import StringType

# Register UDF
def grade(score):
    if score >= 90: return "A"
    elif score >= 70: return "B"
    else: return "C"

grade_udf = udf(grade, StringType())
df.withColumn("grade", grade_udf("score"))

# Pandas UDF (vectorised — faster!)
from pyspark.sql.functions import pandas_udf
import pandas as pd

@pandas_udf(StringType())
def upper_udf(s: pd.Series) -> pd.Series:
    return s.str.upper()

df.withColumn("upper", upper_udf("name"))

⚠️ Avoid regular UDFs when possible — they serialize data through Python, killing Catalyst optimizations. Use Pandas UDFs for large-scale transforms.

Pivot & Unpivot
# Pivot
df.groupBy("dept").pivot("quarter").sum("revenue")

# Pivot with specific values (faster)
df.groupBy("dept").pivot("quarter", ["Q1","Q2","Q3","Q4"]) \
  .agg(F.sum("revenue"))

# Stack / melt (unpivot) — use stack expression
df.selectExpr("id", "stack(2,'Q1',Q1,'Q2',Q2) as (quarter,rev)")

Structured Streaming Basics

      

        
        
# Read from Kafka
stream_df = spark.readStream \
    .format("kafka") \
    .option("kafka.bootstrap.servers", "host:9092") \
    .option("subscribe", "events_topic") \
    .load()

# Read from file (auto-detects new files)
stream_df = spark.readStream \
    .schema(schema) \
    .json("input_dir/")

# Transform stream like a DataFrame!
result = stream_df \
    .groupBy(F.window("timestamp", "10 minutes"), "category") \
    .count()

# Write stream
query = result.writeStream \
    .outputMode("update") \
    .format("console") \
    .trigger(processingTime="1 minute") \
    .start()

query.awaitTermination()

      

    


    

      

        
Output Modes

        

          

            Append

          
+

          

          
Only newly added rows are written. Use for simple transforms without aggregation. Default mode.

        

        

          

            Complete

          
+

          

          
Full result table is rewritten each trigger. Required for aggregations without watermark. Expensive for large datasets.

        

        

          

            Update

          
+

          

          
Only changed rows are written. Efficient for aggregations with watermark. Not supported for sorted output.

df.cache()           # MEMORY_AND_DISK

df.persist(StorageLevel.OFF_HEAP)
